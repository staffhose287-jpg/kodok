<script>
  const canvas = document.getElementById('particles');
  const ctx = canvas.getContext('2d');
  let matrixColor = '#ff003c';
  const fontSize = 16;
  let drops = [];

  function resize() {
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
    const columns = Math.floor(canvas.width / fontSize);
    drops = new Array(columns).fill(1);
  }

  window.onresize = resize;
  window.onload = () => {
    resize();
    if (sessionStorage.getItem('logged') === 'true') {
      document.getElementById('overlay').classList.add('hidden');
      dropTitle();
    }
  };

  function changeTheme(el, color) {
    matrixColor = color;
    const frame = document.querySelector('.outer-frame');
    frame.style.boxShadow = `0 0 60px ${color}`;
    frame.style.background = color;
  }

  function resetTheme() {
    matrixColor = '#ff003c';
    const frame = document.querySelector('.outer-frame');
    frame.style.boxShadow = `0 0 40px rgba(255, 0, 60, 0.3)`;
    frame.style.background = `linear-gradient(135deg, var(--primary), var(--secondary))`;
  }

  function dropTitle() {
    const container = document.getElementById("rain-title");
    container.innerHTML = "";
    "ACIL SHORTCUT".split("").forEach((char, i) => {
      const span = document.createElement("span");
      span.innerText = char === " " ? "\u00A0" : char;
      span.style.animationDelay = `${(i * 0.1).toFixed(2)}s`;
      container.appendChild(span);
    });
  }

  function auth() {
    const u = document.getElementById('uid').value;
    const p = document.getElementById('pass').value;

    const formData = new FormData();
    formData.append('uid', u);
    formData.append('pass', p);

    fetch('login.php', {
      method: 'POST',
      body: formData
    })
    .then(response => response.text())
    .then(data => {
      if (data.trim() === "SUCCESS") {
        document.getElementById('overlay').classList.add('hidden');
        sessionStorage.setItem('logged', 'true');
        dropTitle();
      } else {
        alert("KAMU TIDAK BISA LOGIN LAGI YA HAHA!!");
      }
    })
    .catch(err => {
      console.error("Error:", err);
      alert("Gagal terhubung ke server.");
    });
  }

  function exit() {
    sessionStorage.clear();
    location.reload();
  }

  function draw() {
    ctx.fillStyle = 'rgba(0, 0, 0, 0.15)';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    ctx.fillStyle = matrixColor;
    ctx.font = `900 ${fontSize}px monospace`;
    drops.forEach((y, i) => {
      const text = Math.floor(Math.random() * 2);
      ctx.fillText(text, i * fontSize, y * fontSize);
      if (y * fontSize > canvas.height && Math.random() > 0.98) drops[i] = 0;
      drops[i]++;
    });
  }

  setInterval(draw, 35);
</script>
